STEP 03
Add Apps Script
Go to Extensions → Apps Script. Delete existing code. Paste: 
const SHEET_ID = 'PASTE_YOUR_SHEET_ID_HERE';
function doGet(e){
  const p=e.parameter;
  const ss=SpreadsheetApp.openById(SHEET_ID);
  if(p.action==='read'){
    const sh=ss.getSheetByName(p.tab);
    if(!sh) return j({error:'Tab not found: '+p.tab});
    return j({data:sh.getDataRange().getValues()});
  }
  if(p.action==='append'){
    const sh=ss.getSheetByName(p.tab);
    if(!sh) return j({error:'Tab not found: '+p.tab});
    sh.appendRow(JSON.parse(p.row));
    return j({ok:true});
  }
  return j({error:'unknown action'});
}
function doPost(e){return doGet(e);}
function j(o){
  return ContentService
    .createTextOutput(JSON.stringify(o))
    .setMimeType(ContentService.MimeType.JSON);
}
Replace PASTE_YOUR_SHEET_ID_HERE with the ID from your Sheet URL (between /d/ and /edit).
